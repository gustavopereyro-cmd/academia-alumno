/** * ============================================================ * CLASSROOM DEL ALUMNO + CATÁLOGO PÚBLICO - ACADEMIA VIRTUAL * Backend (Google Apps Script) — Versión con sesiones por token * ============================================================ * ACCIONES DISPONIBLES: * - loginAlumno → login de alumnos (devuelve un token) * - getCursosAlumno → cursos + clases + materiales del alumno * (requiere token; el alumno sale del token) * - GET ?action=catalogo → catálogo público liviano (sin login) * - GET ?action=detalle → Descripcion_Larga de un curso publicado * - GET ?action=limpiarCache&key=... → limpia el caché (requiere clave) * * INSTRUCCIONES: * 1. 999999 : oa; const nb = isNaN(ob) ? 999999 : ob; return na - nb; }); } return cursos.slice().sort((a, b) => String(a.ID_Curso || '').localeCompare(String(b.ID_Curso || '')) ); } function respond(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /* ============================================================ SEGURIDAD: tokens firmados + límite de intentos ============================================================ */ /** Clave secreta para firmar tokens. Se crea sola la primera vez. */ function getSecreto() { const props = PropertiesService.getScriptProperties(); let secreto = props.getProperty('SECRETO_TOKEN'); if (secreto) return secreto; const lock = LockService.getScriptLock(); lock.waitLock(10000); try { secreto = props.getProperty('SECRETO_TOKEN'); if (!secreto) { secreto = Utilities.getUuid() + Utilities.getUuid() + Utilities.getUuid(); props.setProperty('SECRETO_TOKEN', secreto); } } finally { lock.releaseLock(); } return secreto; } function firmar(texto) { return Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(texto, getSecreto())); } function igualesSeguro(a, b) { a = String(a); b = String(b); if (a.length !== b.length) return false; let r = 0; for (let i = 0; i < a.length; i++) r |= a.charCodeAt(i) ^ b.charCodeAt(i); return r === 0; } function crearToken(datos, duracionSeg) { const payload = Object.assign({}, datos, { exp: Date.now() + duracionSeg * 1000 }); const p = Utilities.base64EncodeWebSafe(JSON.stringify(payload), Utilities.Charset.UTF_8); return p + '.' + firmar(p); } /** Devuelve los datos del token si es válido y no venció; si no, null. */ function leerToken(token) { if (!token || typeof token !== 'string') return null; const partes = token.split('.'); if (partes.length !== 2) return null; if (!igualesSeguro(firmar(partes[0]), partes[1])) return null; try { const bytes = Utilities.base64DecodeWebSafe(partes[0]); const payload = JSON.parse(Utilities.newBlob(bytes).getDataAsString('UTF-8')); if (!payload.exp || Date.now() > payload.exp) return null; return payload; } catch (e) { return null; } } function loginBloqueado(clave) { return Number(CacheService.getScriptCache().get('fallos_' + clave) || 0) >= MAX_INTENTOS_LOGIN; } function registrarFallo(clave) { const cache = CacheService.getScriptCache(); const n = Number(cache.get('fallos_' + clave) || 0) + 1; cache.put('fallos_' + clave, String(n), BLOQUEO_LOGIN_SEG); } function limpiarFallos(clave) { CacheService.getScriptCache().remove('fallos_' + clave); } function loginAlumno(emailIngresado, passwordIngresada) { const email = String(emailIngresado || '').trim().toLowerCase(); const password = String(passwordIngresada || '').trim(); const clave = 'alumno_' + email.slice(0, 100); } /* ============================================================ CATÁLOGO ============================================================ */ /** Un curso está publicado si hoy cae dentro de sus fechas de publicación. */ function estaPublicado(c, hoy) { if (c.Fecha_Inicio_Publicacion) { const fi = new Date(c.Fecha_Inicio_Publicacion); } /** * Devuelve el catálogo público (versión LIVIANA, sin Descripcion_Larga). * Usa caché de Apps Script por CACHE_TTL_SECONDS segundos. */ function getCatalogoPublico() { const cache = CacheService.getScriptCache(); try { cache.put(CACHE_KEY_CATALOGO, JSON.stringify(catalogo), CACHE_TTL_SECONDS); } catch (e) { console.warn('No se pudo cachear el catálogo: ' + e.message); } return catalogo; } /** Devuelve la Descripcion_Larga de un curso PUBLICADO. Se pide desde "Ver más". */ function getCursoDetalle(idCurso) { const curso = sheetToObjects('Cursos').find(c => String(c.ID_Curso) === String(idCurso)); if (!curso || !estaPublicado(curso, new Date())) return { error: 'Curso no encontrado' }; return { Descripcion_Larga: curso.Descripcion_Larga || '' }; } /* ============================================================ CLASSROOM DEL ALUMNO ============================================================ */ function getCursosAlumno(idAlumnoDelToken) { const idAlumno = String(idAlumnoDelToken); const inscripciones = sheetToObjects('Inscripciones').filter(i => String(i.ID_Alumno) === idAlumno && String(i.Habilitado || '').trim().toLowerCase() === 'si' ); const idsCursosHabilitados = inscripciones.map(i => String(i.ID_Curso)); const cursosCompletos = cursos.map(curso => { const clasesDelCurso = clases .filter(cl => String(cl.ID_Curso) === String(curso.ID_Curso)) .sort((a, b) => (Number(a.Orden) || 0) - (Number(b.Orden) || 0)) .map(cl => { const contenidos = materiales .filter(m => String(m.ID_Clase) === String(cl.ID_Clase)) .sort((a, b) => (Number(a.Orden) || 0) - (Number(b.Orden) || 0)) .map(m => ({ Tipo: m.Tipo, Nombre: m.Nombre, Link: m.Link, Orden: m.Orden })); }); return { cursos: cursosCompletos }; } /* ============================================================ doGet / doPost ============================================================ */ function doGet(e) { try { const action = e && e.parameter && e.parameter.action;
